% References: write \cref{label}, or \Cref{label} at the start of a sentence. cleveref adds the
% name ("Sec. 3", "Eq. (4)") from the type of the label; the names below set the style for
% the whole paper, e.g., {Section}{Sections} instead of {Sec.}{Secs.}. Load after hyperref.
\usepackage[capitalise,noabbrev]{cleveref}
% Abbreviated within a sentence (\cref), written out at its start (\Cref).

\crefname{subsection}{Sec.}{Secs.}
\crefname{subsubsection}{Sec.}{Secs.}
\crefname{table}{Tab.}{Tabs.}
\crefname{figure}{Figure}{Figures}
\crefname{equation}{Eq.}{Eqs.}
\Crefname{section}{Section}{Sections}
\Crefname{subsection}{Section}{Sections}
\Crefname{subsubsection}{Section}{Sections}
\Crefname{table}{Table}{Tables}

\Crefname{equation}{Equation}{Equations}
